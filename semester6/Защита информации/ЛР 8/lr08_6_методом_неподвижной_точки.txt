> # Атака методом неподвижной точки
> # Выполнила Кочеткова М.П.
> with(RandomTools):
> with(numtheory):
> 
> # Открытый ключ
> e := nextprime(Generate(integer(range = 1 .. 10^3)));
> 
> p := nextprime(Generate(integer(range = 1 .. 10^5))):
> q := nextprime(Generate(integer(range = 1 .. 10^5))):
> n := p * q;
> 
> m := Generate(integer(range = 1 .. n)):
> 
> # Шифртекст
> c := m &^ e mod n;
> 
> # Подбор неподвижной точки
> k := 1;
> c_e := c &^ e mod n:
> res := c_e:
> 
> while res <> c do
>     res := (res * c_e) mod n:
>     k := k + 1:
> end do:
> 
> k := k; # Количество перемножений, порядок
> c_res := c &^ k mod n;
>
> # Проверка
> is(c_res = m);
