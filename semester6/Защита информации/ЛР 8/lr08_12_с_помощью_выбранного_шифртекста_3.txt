> # Атака с помощью выбранного шифртекста 3
> # Выполнила Кочеткова М.П.
> 
> with(RandomTools):
> with(numtheory):
> 
> e := nextprime(Generate(integer(range = 1 .. 10^10)));
> 
> p := nextprime(Generate(integer(range = 1 .. 10^15))):
> q := nextprime(Generate(integer(range = 1 .. 10^15))):
> n := p * q:
> 
> d1 := e &^ (phi(phi(n))-1) mod phi(n);
> 
> m1 := Generate(integer(range = 1 .. 10^15)) mod n;
> m2 := Generate(integer(range = 1 .. 10^15)) mod n;
> m3 := (m1 * m2) mod n;
> 
> res1 := ((m1 &^ d1 mod n) * (m2 &^ d1 mod n)) mod n;
> res2 := m3 &^ d1 mod n;
> 
> # Проверка
> is(res1=res2);
