> # Атака с малой экспонентой (второй вариант)
> # Выполнила Кочеткова М.П.
> with(RandomTools):
> with(numtheory):
> 
> e := 3;
> 
> p1 := nextprime(Generate(integer(range = 1 .. 10^10))):
> q1 := nextprime(Generate(integer(range = 1 .. 10^10))):
> n1 := p1 * q1;
> 
> p2 := nextprime(Generate(integer(range = 1 .. 10^10))):
> q2 := nextprime(Generate(integer(range = 1 .. 10^10))):
> n2 := p2 * q2;
> 
> p3 := nextprime(Generate(integer(range = 1 .. 10^10))):
> q3 := nextprime(Generate(integer(range = 1 .. 10^10))):
> n3 := p3 * q3;
> 
> igcd(e,n1); igcd(e,n2); igcd(e,n3);
> igcd(n1,n2); igcd(n2,n3); igcd(n1,n3);
> 
> m := (((Generate(integer(range = 1 .. n1)) mod n1) mod n2) mod n3);
> 
> C1 := m &^ e mod n1;
> C2 := m &^ e mod n2;
> C3 := m &^ e mod n3;
> 
> m_e := chrem([C1, C2, C3], [n1, n2, n3]);
> 
> m_res := m_e ^ (1/e);
> 
> # Проверка
> is(m = m_res);
